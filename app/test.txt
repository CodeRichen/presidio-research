class PrivacyBridge:
    def __init__(self):
        self.analyzer = AnalyzerEngine()
        self.anonymizer = AnonymizerEngine()
        self.mapping = {}  # 儲存 {占位符: 原始文字}

    def _get_dynamic_operator(self, identifier):
        """建立一個閉包，用來動態生成唯一 ID 並存入 mapping"""
        def replace_with_id(original_value):
            # 建立唯一 ID，例如 [ENTITY_0]
            placeholder = f"[ENTITY_{len(self.mapping)}]"
            self.mapping[placeholder] = original_value
            return placeholder
        return replace_with_id

    def protect_and_ask(self, text):
        # A. 分析文本
        results = self.analyzer.analyze(text=text, language='en')
        
        # B. 執行去識別化，將所有發現的實體都換成動態 ID
        # 注意：我們使用自定義函數來處理所有偵測到的實體
        items_to_replace = sorted(results, key=lambda x: x.start, reverse=True)
        safe_text = text
        
        for res in items_to_replace:
            original_val = text[res.start:res.end]
            # 檢查是否已經存過相同的原始文字，避免重複生成 ID
            existing_id = next((k for k, v in self.mapping.items() if v == original_val), None)
            
            if existing_id:
                placeholder = existing_id
            else:
                placeholder = f"[ENTITY_{len(self.mapping)}]"
                self.mapping[placeholder] = original_val
            
            safe_text = safe_text[:res.start] + placeholder + safe_text[res.end:]

        print(f"--- [步驟 1] 傳送給 Gemini 的安全文本 ---")
        print(safe_text)
        print(f"--- [當前 Mapping] --- \n{self.mapping}\n")

        # C. 詢問 Gemini
        response = model.generate_content(f"Please analyze or reply to this message: {safe_text}")
        ai_reply = response.text

        # D. 還原文字
        print(f"--- [步驟 2] Gemini 原始回覆 ---")
        print(ai_reply)
        
        final_output = ai_reply
        for placeholder, original_val in self.mapping.items():
            final_output = final_output.replace(placeholder, original_val)
            
        return final_output
